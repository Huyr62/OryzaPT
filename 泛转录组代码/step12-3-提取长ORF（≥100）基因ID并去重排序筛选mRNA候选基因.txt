awk '
{
    gene_id = $1
    orf_length = $3

    # 如果该 ORF 长度大于等于 100，记录基因 ID
    if (orf_length >= 100) {
        genes[gene_id] = 1  # 使用基因 ID 作为键，并赋值为 1 (表示该基因至少有一个 ORF 长度 >= 100)

}

    # 遍历所有基因 ID

        print gene_id  # 输出去重后的基因 ID
    }
